    Metodo Estatico Publico main(args: Arreglo de Cadena)
        entrada = nuevo LectorConsola()

        Imprimir("--- SIMULADOR DE SISTEMA HOSPITALARIO ---")
        ImprimirSinSalto("Nombre del hospital: ")
        nombreHospital = eliminarEspacios(entrada.leerLinea())

        Si esVacio(nombreHospital) Entonces
            nombreHospital = "UNAM"
        FinSi

        sistema = nuevo Sistema(nombreHospital)

        opcion = 0
        Hacer
            mostrarMenu()
            opcion = leerEntero(entrada, "Seleccione una opcion: ")

            Segun opcion Hacer
                Caso 1:
                    registrarMedico(entrada)
                Caso 2:
                    registrarEnfermero(entrada)
                Caso 3:
                    registrarPaciente(entrada)
                Caso 4:
                    asignarPacienteMedico(entrada)
                Caso 5:
                    asignarPacienteEnfermero(entrada)
                Caso 6:
                    solicitarConsulta(entrada)
                Caso 7:
                    darTratamientoMedico(entrada)
                Caso 8:
                    darTratamientoEnfermero(entrada)
                Caso 9:
                    verListaMedico(entrada)
                Caso 10:
                    verListaEnfermero(entrada)
                Caso 11:
                    verEstadoPaciente(entrada)
                Caso 12:
                    supervisionJefeEnfermeria(entrada)
                Caso 13:
                    expedirCertificadoSalud(entrada)
                Caso 14:
                    mostrarResumenYPersonalAtencion()
                Caso 0:
                    Imprimir("Sistema hospitalario cerrado correctamente.")
                DeOtroModo:
                    Imprimir("Opcion no valida. Intente de nuevo.")
            FinSegun

            Si opcion != 0 Entonces
                Imprimir("")
            FinSi
        Mientras opcion != 0

        entrada.cerrar()
    FinMetodo

    Metodo Estatico Privado registrarMedico(entrada: LectorConsola)
        Imprimir("-- Registro de Medico --")
        Imprimir("1. Medico General")
        Imprimir("2. Medico Especialista")
        tipo = leerEntero(entrada, "Seleccione el tipo de medico: ")

        nombre = leerTexto(entrada, "Apellidos y nombres: ")
        cedula = leerTexto(entrada, "Cedula profesional: ")

        medico = Nulo
        Si tipo == 1 Entonces
            consultorio = leerEntero(entrada, "Numero de consultorio: ")
            medico = nuevo MedicoGeneral(nombre, cedula, consultorio)
        SiNo
            Si tipo == 2 Entonces
                especialidad = leerTexto(entrada, "Especialidad medica: ")
                cedulaEsp = leerTexto(entrada, "Cedula de especialidad: ")
                hospitalCert = leerTexto(entrada, "Hospital o institucion certificadora: ")
                medico = nuevo MedicoEspecialista(nombre, cedula, especialidad, cedulaEsp, hospitalCert)
            SiNo
                Imprimir("Aviso: Tipo de medico no valido.")
                Retornar
            FinSi
        FinSi

        Si medico.registroEnSistema(sistema) Entonces
            Imprimir("Medico registrado exitosamente.")
        SiNo
            Imprimir("No fue posible registrar al medico.")
        FinSi
    FinMetodo

    Metodo Estatico Privado asignarPacienteMedico(entrada: LectorConsola)
        Imprimir("-- Asignacion de Paciente a Medico --")
        paciente = buscarPacientePorNombre(entrada)
        Si paciente == Nulo Entonces
            Retornar
        FinSi

        medico = buscarMedicoPorCedula(entrada)
        Si medico == Nulo Entonces
            Retornar
        FinSi

        Si sistema.asignarPaciente(paciente, medico) Entonces
            Imprimir("Paciente asignado exitosamente al medico " + medico.getNombre() + ".")
        SiNo
            Imprimir("No fue posible asignar al paciente con el medico.")
        FinSi
    FinMetodo

    Metodo Estatico Privado verEstadoPaciente(entrada: LectorConsola)
        Imprimir("-- Estado Clinico del Paciente --")
        paciente = buscarPacientePorNombre(entrada)
        Si paciente == Nulo Entonces
            Retornar
        FinSi

        Imprimir(paciente.toString())
        Si paciente.estaEnConsulta() Entonces
            Imprimir("Estado: El paciente esta actualmente en consulta.")
        SiNo
            Si paciente.estaEnTratamiento() Entonces
                Imprimir("Estado: El paciente se encuentra en tratamiento.")
            SiNo
                Imprimir("Estado: El paciente esta en espera (PENDIENTE).")
            FinSi
        FinSi
    FinMetodo

    Metodo Estatico Privado supervisionJefeEnfermeria(entrada: LectorConsola)
        Imprimir("-- Modulo de Supervision de Jefatura de Enfermeria --")
        enfermero = buscarEnfermeroPorCedula(entrada)
        Si enfermero == Nulo Entonces
            Retornar
        FinSi

        Si enfermero EsInstanciaDe JefeEnfermeria Entonces
            jefe = (JefeEnfermeria) enfermero
            Imprimir(jefe.supervisarPersonal(sistema.getEnfermeros()))
        SiNo
            Imprimir("Aviso: El enfermero seleccionado no es Jefe de Enfermeria.")
        FinSi
    FinMetodo

    Metodo Estatico Privado expedirCertificadoSalud(entrada: LectorConsola)
        Imprimir("-- Expedicion de Certificado de Salud --")
        medico = buscarMedicoPorCedula(entrada)
        Si medico == Nulo Entonces
            Retornar
        FinSi

        Si NO (medico EsInstanciaDe MedicoGeneral) Entonces
            Imprimir("Aviso: El certificado de salud solo puede ser expedido por un Medico General.")
            Retornar
        FinSi

        paciente = buscarPacientePorNombre(entrada)
        Si paciente == Nulo Entonces
            Retornar
        FinSi

        Si NO (paciente EsInstanciaDe PacienteGeneral) Entonces
            Imprimir("Aviso: El certificado de salud solo puede ser expedido a un Paciente General.")
            Retornar
        FinSi

        mg = (MedicoGeneral) medico
        pg = (PacienteGeneral) paciente
        mg.expedirCertificadoSalud(pg)
    FinMetodo

    Metodo Estatico Privado mostrarResumenYPersonalAtencion()
        Imprimir("-- Resumen Global del Sistema --")
        Imprimir("Hospital: " + sistema.getNombreHospital())
        Imprimir("Medicos registrados: " + sistema.getNoMedicos())
        Imprimir("Enfermeros registrados: " + sistema.getNoEnfermeros())
        Imprimir("Pacientes registrados: " + sistema.getNoPacientes())
        Imprimir("\n-- Catalogo de Atencion Medica --")

        personal = sistema.getPersonalAtencion()
        Si longitud(personal) == 0 Entonces
            Imprimir("No hay personal de atencion registrado actualmente.")
        SiNo
            Para i Desde 0 Hasta longitud(personal) - 1 Hacer
                miembro = personal[i]
                ImprimirFormato("[%02d] %s\n", i + 1, miembro.getDescripcionAtencion())
            FinPara
        FinSi
    FinMetodo

    Metodo Estatico Privado buscarMedicoPorCedula(entrada: LectorConsola): Medico
        cedula = leerTexto(entrada, "Cedula del medico: ")
        medico = sistema.buscarMedico(cedula)
        Si medico == Nulo Entonces
            Imprimir("Aviso: No se encontro ningun medico con la cedula: " + cedula)
        FinSi
        Retornar medico
    FinMetodo

    Metodo Estatico Privado leerTexto(entrada: LectorConsola, mensaje: Cadena): Cadena
        ImprimirSinSalto(mensaje)
        valor = eliminarEspacios(entrada.leerLinea())
        Retornar valor
    FinMetodo

    Metodo Estatico Privado leerEntero(entrada: LectorConsola, mensaje: Cadena): Entero
        ImprimirSinSalto(mensaje)
        Intentar
            Retornar convertirAEntero(eliminarEspacios(entrada.leerLinea()))
        Capturar ErrorFormatoNumero
            Retornar -1
        FinIntentar
    FinMetodo
