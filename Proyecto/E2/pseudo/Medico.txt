ClaseAbstracta Medico Implementa AtencionMedica
    Constante Publico CAPACIDAD_MAXIMA_PACIENTES = 10

    Atributos:
        Privado nombre: Cadena
        Privado cedula: Cadena
        Privado especialidad: Cadena
        Privado pacientes: Lista de Paciente
        Privado pacienteEnConsulta: Paciente
        Privado sistema: Sistema

    Constructor(nombre: Cadena, cedula: Cadena, especialidad: Cadena)
        Si nombre == Nulo O esVacio(eliminarEspacios(nombre)) Entonces
            this.nombre = "Sin asignar"
        SiNo
            this.nombre = eliminarEspacios(nombre)
        FinSi

        Si cedula == Nulo O esVacio(eliminarEspacios(cedula)) Entonces
            this.cedula = "Sin cedula"
        SiNo
            this.cedula = eliminarEspacios(cedula)
        FinSi

        Si especialidad == Nulo O esVacio(eliminarEspacios(especialidad)) Entonces
            this.especialidad = "General"
        SiNo
            this.especialidad = eliminarEspacios(especialidad)
        FinSi

        this.pacientes = nueva Lista()
        this.pacienteEnConsulta = Nulo
    FinConstructor

    Metodo registroEnSistema(sistema: Sistema): Booleano
        Si sistema == Nulo Entonces
            Imprimir("Aviso: El sistema no puede ser nulo.")
            Retornar Falso
        FinSi

        Si this.sistema == sistema Entonces
            Retornar Verdadero
        FinSi

        Si sistema.registroMedico(this) Entonces
            this.sistema = sistema
            Retornar Verdadero
        SiNo
            Retornar Falso
        FinSi
    FinMetodo

    Metodo solicitarPaciente(): Paciente
        Para Cada p En this.pacientes Hacer
            Si p != Nulo Y p.estaPendiente() Entonces
                Retornar p
            FinSi
        FinPara
        Retornar Nulo
    FinMetodo

    MetodoAbstracto darConsulta(paciente: Paciente): Booleano

    Metodo darTratamiento(paciente: Paciente): Booleano
        Si paciente == Nulo Entonces
            Imprimir("Aviso: El paciente no puede ser nulo.")
            Retornar Falso
        FinSi

        Si getPacienteAsignado(paciente) == Nulo Entonces
            Imprimir("Aviso: El paciente no esta asignado a este medico.")
            Retornar Falso
        FinSi

        paciente.setEstado(Paciente.ESTADO_EN_TRATAMIENTO)
        Si this.pacienteEnConsulta == paciente Entonces
            this.pacienteEnConsulta = Nulo
        FinSi
        Retornar Verdadero
    FinMetodo

    Metodo liberarConsulta(): Booleano
        Si this.pacienteEnConsulta != Nulo Entonces
            this.pacienteEnConsulta = Nulo
            Retornar Verdadero
        SiNo
            Retornar Falso
        FinSi
    FinMetodo

    Metodo verListaPacientes(): Cadena
        Si longitud(this.pacientes) == 0 Entonces
            Retornar ""
        FinSi

        copia = nueva Lista(this.pacientes)
        Para i Desde 0 Hasta longitud(copia) - 2 Hacer
            Para j Desde 0 Hasta longitud(copia) - 2 - i Hacer
                Si compararIgnorandoMayusculas(copia[j].getNombre(), copia[j + 1].getNombre()) < 0 Entonces
                    temporal = copia[j]
                    copia[j] = copia[j + 1]
                    copia[j + 1] = temporal
                FinSi
            FinPara
        FinPara

        sb = ""
        Para i Desde 0 Hasta longitud(copia) - 1 Hacer
            sb = sb + copia[i].getNombre() + " - " + copia[i].getEspecialidadAtencion()
            Si i < longitud(copia) - 1 Entonces
                sb = sb + "\n"
            FinSi
        FinPara
        Retornar sb
    FinMetodo

    Metodo agregarPaciente(paciente: Paciente): Booleano
        Si paciente == Nulo Entonces
            Imprimir("Aviso: El paciente no puede ser nulo.")
            Retornar Falso
        FinSi

        Si longitud(this.pacientes) >= CAPACIDAD_MAXIMA_PACIENTES Entonces
            Imprimir("Aviso: El medico ha alcanzado su capacidad maxima (10 pacientes).")
            Retornar Falso
        FinSi

        Si NO igualIgnorandoMayusculas(this.especialidad, paciente.getEspecialidadAtencion()) Entonces
            Imprimir("Aviso: La especialidad requerida por el paciente no coincide con la del medico.")
            Retornar Falso
        FinSi

        Si getPacienteAsignado(paciente) != Nulo Entonces
            Imprimir("Aviso: El paciente ya se encuentra asignado a este medico.")
            Retornar Falso
        FinSi

        this.pacientes.agregar(paciente)
        paciente.setMedicoAsignado(this)
        Retornar Verdadero
    FinMetodo

    Metodo getPacienteAsignado(paciente: Paciente): Paciente
        Si paciente == Nulo Entonces
            Retornar Nulo
        FinSi

        Si contiene(this.pacientes, paciente) Entonces
            Retornar paciente
        FinSi
        Retornar Nulo
    FinMetodo

    Metodo getNombre(): Cadena
        Retornar this.nombre
    FinMetodo

    Metodo setNombre(nombre: Cadena)
        Si nombre == Nulo O esVacio(eliminarEspacios(nombre)) Entonces
            Imprimir("Aviso: El nombre no puede estar vacio.")
        SiNo
            this.nombre = eliminarEspacios(nombre)
        FinSi
    FinMetodo

    Metodo getCedula(): Cadena
        Retornar this.cedula
    FinMetodo

    Metodo getEspecialidad(): Cadena
        Retornar this.especialidad
    FinMetodo

    Metodo setEspecialidad(especialidad: Cadena)
        Si especialidad == Nulo O esVacio(eliminarEspacios(especialidad)) Entonces
            Imprimir("Aviso: La especialidad no puede estar vacia.")
        SiNo
            this.especialidad = eliminarEspacios(especialidad)
        FinSi
    FinMetodo

    Metodo getNoPacientes(): Entero
        Retornar longitud(this.pacientes)
    FinMetodo

    Metodo getPacienteEnConsulta(): Paciente
        Retornar this.pacienteEnConsulta
    FinMetodo

    Metodo Protegido setPacienteEnConsulta(paciente: Paciente)
        this.pacienteEnConsulta = paciente
    FinMetodo

    Metodo getPacientes(): Lista de Paciente
        Retornar nueva Lista(this.pacientes)
    FinMetodo

    Metodo getSistema(): Sistema
        Retornar this.sistema
    FinMetodo

    Metodo setSistema(sistema: Sistema)
        this.sistema = sistema
    FinMetodo

    Metodo toString(): Cadena
        enConsulta = "Ninguno"
        Si pacienteEnConsulta != Nulo Entonces
            enConsulta = pacienteEnConsulta.getNombre()
        FinSi

        Retornar "Medico: " + this.nombre + "\nCedula: " + this.cedula + "\nEspecialidad: " + this.especialidad + "\nPacientes a cargo: " + longitud(this.pacientes) + "/" + CAPACIDAD_MAXIMA_PACIENTES + "\nEn consulta activa: " + enConsulta
    FinMetodo
FinClase
