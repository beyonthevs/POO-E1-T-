    Constructor(nombreHospital: Cadena)
        Si nombreHospital == Nulo O esVacio(eliminarEspacios(nombreHospital)) Entonces
            this.nombreHospital = "UNAM"
        SiNo
            this.nombreHospital = eliminarEspacios(nombreHospital)
        FinSi

        this.medicos = nueva Lista()
        this.enfermeros = nueva Lista()
        this.pacientes = nueva Lista()
        this.medicosPorCedula = nuevo Diccionario()
        this.enfermerosPorCedula = nuevo Diccionario()
        this.pacientesPorNombre = nuevo Diccionario()
    FinConstructor

    Metodo registroMedico(medico: Medico): Booleano
        Si medico == Nulo Entonces
            Imprimir("Aviso: El medico no puede ser nulo.")
            Retornar Falso
        FinSi

        clave = mayusculas(eliminarEspacios(medico.getCedula()))
        Si contieneClave(this.medicosPorCedula, clave) Entonces
            Imprimir("Aviso: Ya existe un medico registrado con la cedula: " + medico.getCedula())
            Retornar Falso
        FinSi

        agregar(this.medicos, medico)
        poner(this.medicosPorCedula, clave, medico)
        Si medico.getSistema() != this Entonces
            medico.setSistema(this)
        FinSi
        Retornar Verdadero
    FinMetodo

    Metodo asignarPaciente(paciente: Paciente, medico: Medico): Booleano
        Si paciente == Nulo O medico == Nulo Entonces
            Imprimir("Aviso: Paciente o medico nulo en la asignacion.")
            Retornar Falso
        FinSi

        Si NO estaRegistrado(paciente) Entonces
            Imprimir("Aviso: El paciente no se encuentra registrado en el sistema.")
            Retornar Falso
        FinSi

        Si NO estaRegistrado(medico) Entonces
            Imprimir("Aviso: El medico no se encuentra registrado en el sistema.")
            Retornar Falso
        FinSi

        Si paciente EsInstanciaDe PacienteGeneral Entonces
            Si medico EsInstanciaDe MedicoEspecialista Entonces
                Imprimir("Aviso: Un medico especialista no puede recibir pacientes generales.")
                Retornar Falso
            FinSi
            Si NO (medico EsInstanciaDe MedicoGeneral) Entonces
                Imprimir("Aviso: Los pacientes generales solo pueden asignarse a medicos generales.")
                Retornar Falso
            FinSi
        FinSi

        Si paciente EsInstanciaDe PacienteEspecialidad Entonces
            Si NO (medico EsInstanciaDe MedicoEspecialista) Entonces
                Imprimir("Aviso: Los pacientes de especialidad solo pueden asignarse a medicos especialistas.")
                Retornar Falso
            FinSi
            Si NO igualIgnorandoMayusculas(paciente.getEspecialidadAtencion(), medico.getEspecialidad()) Entonces
                Imprimir("Aviso: La especialidad requerida por el paciente no coincide con la del medico.")
                Retornar Falso
            FinSi
        FinSi

        Si NO igualIgnorandoMayusculas(medico.getEspecialidad(), paciente.getEspecialidadAtencion()) Entonces
            Imprimir("Aviso: No coincide la especialidad de atencion.")
            Retornar Falso
        FinSi

        Retornar medico.agregarPaciente(paciente)
    FinMetodo

    FinMetodo
